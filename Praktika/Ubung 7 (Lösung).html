package de.thm.mni.pi2.database;

import java.sql.*;

/**
 * Main-Klasse zur Ausgabe der Studierenden mit ihren Noten
 * Verwendet JDBC mit PreparedStatement für einen JOIN-Query
 */
public class StudentNotenAusgabe {

    private static final String DB_URL = "jdbc:mysql://localhost:3306/pi2";
    private static final String DB_USER = "user1";
    private static final String DB_PASSWORD = "passwort";
    
    public static void main(String[] args) {
        String StudentName="Mueller";
        if(args.length > 0) {
                StudentName = args[0];
        }
        StudentNotenAusgabe studentNotenAusgabe = new StudentNotenAusgabe();

        studentNotenAusgabe.zeigeStudentenMitNoten();
        studentNotenAusgabe.zeigeNotenFuerStudent(StudentName);
    }
    
    /**
     * Zeigt alle Studierenden mit ihren Noten an
     * Verwendet einen JOIN über die Tabellen students, ratings und kurs
     */
    public void zeigeStudentenMitNoten() {
        String sql = "select s.name, k.modulname, r.rating " +
                     "from students s " +
                     "join ratings r on s.matrikelnr=r.sid " +
                     "join kurs k on r.kid=k.modulnr";
        executeQuery(sql, "=== Liste der Studierenden mit ihren Noten ===", null);
    }
    
    /**
     * Zeigt die Noten für einen bestimmten Studierenden an
     * Verwendet einen JOIN mit WHERE-Klausel und PreparedStatement-Parameter
     * @param studentName Name des Studierenden
     */
    public void zeigeNotenFuerStudent(String studentName) {
        String sql = "select s.name, k.modulname, r.rating " +
                     "from students s " +
                     "join ratings r on s.matrikelnr=r.sid " +
                     "join kurs k on r.kid=k.modulnr " +
                     "where s.name = ?";
        executeQuery(sql, "=== Noten für Student: " + studentName + " ===", studentName);
    }
    
    /**
     * Führt eine SQL-Abfrage aus und gibt die Ergebnisse formatiert aus
     * @param sql SQL-Query mit optionalem Parameter (?)
     * @param ueberschrift Überschrift für die Ausgabe
     * @param studentName Optional: Name des Studierenden für WHERE-Klausel
     */
    private void executeQuery(String sql, String ueberschrift, String studentName) {
        try (Connection connection = DriverManager.getConnection(DB_URL, DB_USER, DB_PASSWORD);
             PreparedStatement preparedStatement = connection.prepareStatement(sql)) {
            
            // Parameter setzen, falls vorhanden
            if (studentName != null) {
                preparedStatement.setString(1, studentName);
            }
            
            try (ResultSet resultSet = preparedStatement.executeQuery()) {
                System.out.println("\n" + ueberschrift + "\n");
                System.out.printf("%-20s %-35s %-6s%n", 
                    "Name", "ModulName", "Rating");
                System.out.println("--------------------------------------------------------------------------------");
                
                boolean hasResults = false;
                while (resultSet.next()) {
                    hasResults = true;
                    String name = resultSet.getString("name");
                    String modulName = resultSet.getString("modulname");
                    int rating = resultSet.getInt("rating");
                    
                    System.out.printf("%-20s %-35s %-6d%n",
                        name, modulName, rating);
                }
                
                if (!hasResults) {
                    String fehlermeldung = studentName != null 
                        ? "Keine Daten für Student '" + studentName + "' gefunden."
                        : "Keine Daten gefunden.";
                    System.out.println(fehlermeldung);
                }
            }
            
        } catch (SQLException e) {
            System.err.println("Fehler beim Ausführen der Datenbankabfrage:");
            System.err.println("SQL State: " + e.getSQLState());
            System.err.println("Error Code: " + e.getErrorCode());
            System.err.println("Message: " + e.getMessage());
            e.printStackTrace();
        }
    }
}
