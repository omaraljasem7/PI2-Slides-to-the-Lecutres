package de.thm.mni.pi2.adder;

import io.vertx.core.Future;
import io.vertx.core.VerticleBase;
import io.vertx.core.http.HttpMethod;
import io.vertx.core.json.JsonObject;
import io.vertx.ext.web.Router;
import io.vertx.ext.web.RoutingContext;
import io.vertx.ext.web.handler.BodyHandler;
import io.vertx.ext.web.handler.CorsHandler;
import io.vertx.ext.web.handler.StaticHandler;

import java.time.Month;
import java.time.format.TextStyle;
import java.util.*;

public class AdderVerticle extends VerticleBase {
  int counter = 0;

  @Override
  public Future<?> start() {
    int port = 8888;

    Router router = Router.router(vertx);
    // Allow CORS for all origins
    router.route().handler(CorsHandler.create()
      .allowedMethod(HttpMethod.GET)
      .allowedMethod(HttpMethod.POST)
      .allowedMethod(HttpMethod.PUT)
      .allowedMethod(HttpMethod.DELETE)
      .allowedHeader("Content-Type")
      .allowedHeader("Authorization"));

    router.post("/add")
      .handler(BodyHandler.create()).handler(this::add);

    router.get("/counter").handler(rc -> {
       rc.response()
         .setStatusCode(200)
         .putHeader("content-type", "text/plain")
         .end("Anzahl Aufrufe: " + counter);
    });

    router.route("/static/*").handler(
      StaticHandler.create("public"));

    router.get("/monat/:nummer").handler(rc -> {
      int nummer = Integer.parseInt(rc.request().getParam("nummer"));
      String monat = switch (nummer) {
        case 1 -> "Januar";
        case 2 -> "Februar";
        case 3 -> "März";
        case 4 -> "April";
        case 5 -> "Mai";
        case 6 -> "Juni";
        case 7 -> "Juli";
        case 8 -> "August";
        case 9 -> "September";
        case 10 -> "Oktober";
        case 11 -> "November";
        case 12 -> "Dezember";
        default -> "Invalid Month";
      };

      // Alternativ:
      // monat = Month.of(nummer).getDisplayName(TextStyle.FULL, Locale.GERMAN);
      rc.response().setStatusCode(200).end(monat );
    });

    return vertx
      .createHttpServer()
      .requestHandler(router)
      .listen(port).onSuccess(http -> {
        System.out.println("HTTP server started on port 8888");
      });
  }

  private void add(RoutingContext rc) {
    try {
      counter++;
      JsonObject jsonBody = rc.body().asJsonObject();
      int a = jsonBody.getInteger("a");
      int b = jsonBody.getInteger("b");
      int c = jsonBody.getInteger("c");
      JsonObject ergebnis = new JsonObject().put("result", a + b + c);
      rc.response().setStatusCode(200)
        .putHeader("content-type", "application/json")
        .end(ergebnis.encode());
    } catch (Exception e) { rc.response().setStatusCode(400).end("Ungültiges JSON");  }
  }
}
