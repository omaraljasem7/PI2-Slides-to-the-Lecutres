use pi2;

create table dozent (
    id int primary key,
    Name varchar(20) not null
    );


create table kurs (
    ModulNr varchar(7) not null,
    ModulName text,
    CrP int default 6,
     PRIMARY KEY (ModulNr),
    Dozent_id int NOT NULL,
    FOREIGN KEY (Dozent_id) REFERENCES dozent(id)
    );
