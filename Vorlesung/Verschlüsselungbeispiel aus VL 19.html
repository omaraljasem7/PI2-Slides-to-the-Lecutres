package org.example;

import javax.crypto.BadPaddingException;
import javax.crypto.Cipher;
import java.security.*;
import java.util.Base64;
import java.util.Scanner;
import java.nio.charset.StandardCharsets;

/**
 * Demonstration der doppelten Verschlüsselung zwischen Alice und Bob:
 * 1. Alice verschlüsselt mit ihrem privaten Schlüssel
 * 2. Alice verschlüsselt dann mit Bobs öffentlichem Schlüssel
 * 3. Bob entschlüsselt mit seinem privaten Schlüssel
 * 4. Bob entschlüsselt dann mit Alices öffentlichem Schlüssel
 */
public class AliceBobEncryptionDemo {
    
    // ANSI-Farbcodes für Terminal-Ausgabe
    private static final String RESET = "\033[0m";
    private static final String ALICE_COLOR = "\033[35m"; // Magenta für Alice
    private static final String BOB_COLOR = "\033[36m";   // Cyan für Bob
    private static final String MITM_COLOR = "\033[33m";  // Gelb für Man-in-the-Middle
    
    public static void main(String[] args) {
        try {
            String originalMessage = "Hallo Bob, dies ist eine geheime Nachricht!";
            
            // Schlüsselpaare generieren
            printAlice("Generiere Schlüsselpaar für Alice...");
            KeyPair aliceKeyPair = generateKeyPair();
            printAlice("✓ Schlüsselpaar für Alice generiert!");
            
            printBob("Generiere Schlüsselpaar für Bob...");
            KeyPair bobKeyPair = generateKeyPair();
            printBob("✓ Schlüsselpaar für Bob generiert!");
            System.out.println();
            
            // Schritt 1: Alice verschlüsselt mit ihrem privaten Schlüssel
            // HINWEIS: In der Praxis würde man hier signieren (nicht verschlüsseln).
            // Dies ist nur für Demonstrationszwecke der doppelten Verschlüsselung.
            printAlice("Schritt 1: Verschlüssele Nachricht mit meinem privaten Schlüssel...");
            printAlice("Meine Nachricht ist: " + originalMessage);
            byte[] encryptedWithAlicePrivate = encrypt(originalMessage, aliceKeyPair.getPrivate());
            printAlice("✓ Nachricht mit privatem Schlüssel verschlüsselt");
            printAlice("Verschlüsselte Daten: " + Base64.getEncoder().encodeToString(encryptedWithAlicePrivate));
            System.out.println();
            
            // Schritt 2: Alice verschlüsselt mit Bobs öffentlichem Schlüssel
            printAlice("Schritt 2: Verschlüssele erneut mit Bobs öffentlichem Schlüssel...");
            byte[] encryptedWithBobPublic = encrypt(encryptedWithAlicePrivate, bobKeyPair.getPublic());
            printAlice("✓ Nachricht mit Bobs öffentlichem Schlüssel verschlüsselt");
            printAlice("Doppelt verschlüsselte Daten: " + Base64.getEncoder().encodeToString(encryptedWithBobPublic));
            printAlice("→ Sende verschlüsselte Nachricht an Bob...");
            System.out.println();
            
            // Man-in-the-Middle Simulation
            System.out.println(MITM_COLOR + "═══════════════════════════════════════════════════════" + RESET);
            System.out.println(MITM_COLOR + "[MAN-IN-THE-MIDDLE] Nachricht abgefangen!" + RESET);
            System.out.println(MITM_COLOR + "═══════════════════════════════════════════════════════" + RESET);
            
            try (Scanner scanner = new Scanner(System.in)) {
                System.out.print(MITM_COLOR + "[MITM] Möchten Sie die verschlüsselte Nachricht manipulieren? (ja/nein): " + RESET);
                String userInput = scanner.nextLine().trim().toLowerCase();
                
                if (userInput.equals("ja") || userInput.equals("j") || userInput.equals("yes") || userInput.equals("y")) {
                    System.out.println(MITM_COLOR + "[MITM] Manipuliere verschlüsselte Nachricht..." + RESET);
                    
                    // Manipulation: Ändere einige Bytes in der verschlüsselten Nachricht
                    byte[] manipulatedMessage = encryptedWithBobPublic.clone();
                    
                    // Ändere die ersten paar Bytes (simuliert Manipulation)
                    if (manipulatedMessage.length > 10) {
                        for (int i = 0; i < Math.min(10, manipulatedMessage.length); i++) {
                            manipulatedMessage[i] = (byte) (manipulatedMessage[i] ^ 0xFF); // XOR mit 0xFF invertiert die Bits
                        }
                    }
                    
                    encryptedWithBobPublic = manipulatedMessage;
                    System.out.println(MITM_COLOR + "[MITM] ✓ Nachricht wurde manipuliert!" + RESET);
                    System.out.println(MITM_COLOR + "[MITM] Manipulierte Daten: " + Base64.getEncoder().encodeToString(encryptedWithBobPublic) + RESET);
                } else {
                    System.out.println(MITM_COLOR + "[MITM] Nachricht wird unverändert weitergeleitet." + RESET);
                }
            }
            System.out.println(MITM_COLOR + "[MITM] → Leite Nachricht an Bob weiter..." + RESET);
            System.out.println(MITM_COLOR + "═══════════════════════════════════════════════════════" + RESET);
            System.out.println();
            
            // Schritt 3: Bob empfängt und entschlüsselt mit seinem privaten Schlüssel
            printBob("Schritt 3: Empfange Nachricht von Alice...");
            printBob("Entschlüssele mit meinem privaten Schlüssel...");
            byte[] decryptedWithBobPrivate = decrypt(encryptedWithBobPublic, bobKeyPair.getPrivate());
            
            if (decryptedWithBobPrivate == null) {
                printBob("✗ FEHLER: Kann Nachricht nicht entschlüsseln!");
                printBob("✗ Die Nachricht wurde möglicherweise manipuliert oder beschädigt.");
                printBob("✗ BadPaddingException: Entschlüsselung fehlgeschlagen!");
                return;
            }
            
            printBob("✓ Nachricht mit privatem Schlüssel entschlüsselt");
            printBob("Zwischenergebnis: " + Base64.getEncoder().encodeToString(decryptedWithBobPrivate));
            System.out.println();
            
            // Schritt 4: Bob entschlüsselt mit Alices öffentlichem Schlüssel
            printBob("Schritt 4: Entschlüssele mit Alices öffentlichem Schlüssel...");
            byte[] decryptedWithAlicePublic = decrypt(decryptedWithBobPrivate, aliceKeyPair.getPublic());
            
            if (decryptedWithAlicePublic == null) {
                printBob("✗ FEHLER: Kann Nachricht nicht entschlüsseln!");
                printBob("✗ Die Nachricht wurde möglicherweise manipuliert oder beschädigt.");
                printBob("✗ BadPaddingException: Entschlüsselung fehlgeschlagen!");
                return;
            }
            
            String finalMessage = new String(decryptedWithAlicePublic, StandardCharsets.UTF_8);
            printBob("✓ Nachricht mit Alices öffentlichem Schlüssel entschlüsselt");
            System.out.println();
            
            // Ergebnis anzeigen
            printBob("═══════════════════════════════════════════════════════");
            printBob("Original-Nachricht erfolgreich entschlüsselt:");
            printBob(">>> " + finalMessage);
            printBob("═══════════════════════════════════════════════════════");
            
            // Verifikation
            if (originalMessage.equals(finalMessage)) {
                printBob("✓ Verifikation erfolgreich: Nachricht ist identisch!");
            } else {
                printBob("✗ FEHLER: Nachricht stimmt nicht überein!");
            }
            
        } catch (Exception e) {
            System.err.println("Fehler: " + e.getMessage());
            e.printStackTrace();
        }
    }
    
    /**
     * Generiert ein RSA-Schlüsselpaar
     */
    private static KeyPair generateKeyPair() throws NoSuchAlgorithmException {
        KeyPairGenerator keyGen = KeyPairGenerator.getInstance("RSA");
        keyGen.initialize(2048);
        return keyGen.generateKeyPair();
    }
    
    /**
     * Maximale Blockgröße für RSA-Verschlüsselung (2048 Bit = 256 Bytes, minus 11 Bytes für PKCS1Padding)
     */
    private static final int MAX_ENCRYPT_BLOCK_SIZE = 245;
    private static final int CIPHER_BLOCK_SIZE = 256; // Größe eines verschlüsselten Blocks
    
    /**
     * Verschlüsselt Daten mit einem RSA-Schlüssel (unterstützt große Daten durch Chunking)
     */
    private static byte[] encrypt(byte[] data, Key key) throws Exception {
        Cipher cipher = Cipher.getInstance("RSA");
        cipher.init(Cipher.ENCRYPT_MODE, key);
        
        // Wenn die Daten klein genug sind, direkt verschlüsseln
        if (data.length <= MAX_ENCRYPT_BLOCK_SIZE) {
            return cipher.doFinal(data);
        }
        
        // Für größere Daten: Chunking verwenden
        java.io.ByteArrayOutputStream outputStream = new java.io.ByteArrayOutputStream();
        int offset = 0;
        
        while (offset < data.length) {
            int chunkSize = Math.min(MAX_ENCRYPT_BLOCK_SIZE, data.length - offset);
            byte[] chunk = new byte[chunkSize];
            System.arraycopy(data, offset, chunk, 0, chunkSize);
            
            byte[] encryptedChunk = cipher.doFinal(chunk);
            outputStream.write(encryptedChunk);
            
            offset += chunkSize;
        }
        
        return outputStream.toByteArray();
    }
    
    /**
     * Verschlüsselt einen String mit einem RSA-Schlüssel
     */
    private static byte[] encrypt(String data, Key key) throws Exception {
        return encrypt(data.getBytes(), key);
    }
    
    /**
     * Entschlüsselt Daten mit einem RSA-Schlüssel (unterstützt große Daten durch Chunking)
     * Gibt null zurück, wenn BadPaddingException auftritt (z.B. bei manipulierten Daten)
     */
    private static byte[] decrypt(byte[] encryptedData, Key key) {
        try {
            Cipher cipher = Cipher.getInstance("RSA");
            cipher.init(Cipher.DECRYPT_MODE, key);
            
            // Wenn die Daten nur einen Block enthalten, direkt entschlüsseln
            if (encryptedData.length <= CIPHER_BLOCK_SIZE) {
                return cipher.doFinal(encryptedData);
            }
            
            // Für mehrere Blöcke: Chunking verwenden
            java.io.ByteArrayOutputStream outputStream = new java.io.ByteArrayOutputStream();
            int offset = 0;
            
            while (offset < encryptedData.length) {
                int chunkSize = Math.min(CIPHER_BLOCK_SIZE, encryptedData.length - offset);
                byte[] chunk = new byte[chunkSize];
                System.arraycopy(encryptedData, offset, chunk, 0, chunkSize);
                
                byte[] decryptedChunk = cipher.doFinal(chunk);
                outputStream.write(decryptedChunk);
                
                offset += chunkSize;
            }
            
            return outputStream.toByteArray();
        } catch (BadPaddingException e) {
            // BadPaddingException tritt auf, wenn die Daten manipuliert wurden
            // oder der falsche Schlüssel verwendet wurde
            return null;
        } catch (Exception e) {
            // Andere Exceptions weiterwerfen
            throw new RuntimeException("Entschlüsselungsfehler: " + e.getMessage(), e);
        }
    }
    
    /**
     * Gibt eine Nachricht von Alice in Magenta aus
     */
    private static void printAlice(String message) {
        System.out.println(ALICE_COLOR + "[ALICE] " + message + RESET);
    }
    
    /**
     * Gibt eine Nachricht von Bob in Cyan aus
     */
    private static void printBob(String message) {
        System.out.println(BOB_COLOR + "[BOB]   " + message + RESET);
    }
}
